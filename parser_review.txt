import base64
import json
import mimetypes
import re

from config import client, CLASSIFIER_MODEL


DOCUMENT_TYPES = {
    "Prescription",
    "Medicine Strip",
    "Lab Report",
    "Radiology Report",
    "Medical Imaging",
    "Invoice",
    "Discharge Summary",
    "Medical Certificate",
    "Insurance Document",
    "Unknown",
}

STANDARD_TYPES = DOCUMENT_TYPES - {
    "Medical Imaging",
    "Radiology Report",
    "Unknown",
}

VALID_ROUTING_ACTIONS = {
    "store_medical_image",
    "parse_radiology_report",
    "standard_document_parser",
    "manual_review",
}

IMAGING_SUBTYPES = {
    "MRI Scan",
    "CT Scan",
    "X-ray Image",
    "Ultrasound Image",
}


def encode_image(path):
    with open(path, "rb") as file:
        return base64.b64encode(file.read()).decode("utf-8")


def get_image_mime_type(path):
    mime_type, _ = mimetypes.guess_type(path)

    if mime_type and mime_type.startswith("image/"):
        return mime_type

    return "image/jpeg"


SYSTEM_PROMPT = """
You are GoMeds AI, a healthcare document classification engine.

Classify the supplied healthcare document into exactly ONE
of these ten top-level categories:

1. Prescription
2. Medicine Strip
3. Lab Report
4. Radiology Report
5. Medical Imaging
6. Invoice
7. Discharge Summary
8. Medical Certificate
9. Insurance Document
10. Unknown

CATEGORY DEFINITIONS

Prescription:
A doctor's handwritten or printed prescription.

Medicine Strip:
Medicine packaging, blister packs or tablet strips.

Lab Report:
A laboratory report containing test names, measured values,
units, reference ranges or laboratory findings.

Radiology Report:
A written radiology report containing a radiologist's findings,
impression or recommendations.

Medical Imaging:
An actual medical scan or imaging study, including MRI, CT,
X-ray and ultrasound images.

Invoice:
A billing document or invoice.

Discharge Summary:
A hospital discharge summary.

Medical Certificate:
A certificate issued by a healthcare professional.

Insurance Document:
A healthcare-related insurance document.

Unknown:
A document that cannot reliably be classified into another
category above.

IMPORTANT DISTINCTIONS

1. ACTUAL MEDICAL SCAN

If the input is an actual MRI, CT, X-ray or ultrasound image,
classify it as:

- document_type: "Medical Imaging"
- document_subtype: "MRI Scan", "CT Scan",
  "X-ray Image", or "Ultrasound Image"
- routing_action: "store_medical_image"

2. WRITTEN RADIOLOGY REPORT

If the input is a written report containing radiology findings,
an impression or recommendations, classify it as:

- document_type: "Radiology Report"
- document_subtype: "MRI Report", "CT Report",
  "X-ray Report", "Ultrasound Report", or "Radiology Report"
- routing_action: "parse_radiology_report"

3. OTHER RECOGNIZED DOCUMENTS

For Prescription, Medicine Strip, Lab Report, Invoice,
Discharge Summary, Medical Certificate or Insurance Document:

- document_subtype: null
- routing_action: "standard_document_parser"

4. UNCERTAIN DOCUMENT

If the document cannot be classified reliably:

- document_type: "Unknown"
- document_subtype: null
- routing_action: "manual_review"

Return only valid JSON with this structure:

{
    "document_type": "Medical Imaging",
    "document_subtype": "MRI Scan",
    "routing_action": "store_medical_image",
    "confidence": 0.98
}

The values are examples. Classify the actual input.

Confidence must be a number from 0 to 1.

Do not invent medical findings, diagnoses or patient information.
Do not confuse an actual scan with its written report.
Do not include Markdown or explanations outside the JSON.
"""


def classify_document(image_path):
    image = encode_image(image_path)
    mime_type = get_image_mime_type(image_path)

    response = client.chat.completions.create(
        model=CLASSIFIER_MODEL,
        temperature=0,
        max_tokens=1024,
        reasoning_effort="low",
        messages=[
            {
                "role": "system",
                "content": SYSTEM_PROMPT,
            },
            {
                "role": "user",
                "content": [
                    {
                        "type": "text",
                        "text": (
                            "Classify this healthcare document. "
                            "Return only the required JSON."
                        ),
                    },
                    {
                        "type": "image_url",
                        "image_url": {
                            "url": f"data:{mime_type};base64,{image}",
                        },
                    },
                ],
            },
        ],
    )

    choice = response.choices[0]
    content = choice.message.content

    print("\n===== CLASSIFIER RESPONSE =====")
    print("Model:", CLASSIFIER_MODEL)
    print("Finish reason:", choice.finish_reason)

    if not content:
        return {
            "document_type": "Unknown",
            "document_subtype": None,
            "routing_action": "manual_review",
            "confidence": 0.0,
        }

    content = re.sub(
        r"^```(?:json)?\s*|\s*```$",
        "",
        content.strip(),
        flags=re.IGNORECASE,
    ).strip()

    try:
        result = json.loads(content)

    except json.JSONDecodeError:
        start = content.find("{")
        end = content.rfind("}")

        try:
            if start == -1 or end <= start:
                raise ValueError("No JSON object found.")

            result = json.loads(content[start:end + 1])

        except (json.JSONDecodeError, ValueError):
            print("Invalid classifier JSON.")

            return {
                "document_type": "Unknown",
                "document_subtype": None,
                "routing_action": "manual_review",
                "confidence": 0.0,
                "raw": content,
            }

    if not isinstance(result, dict):
        return {
            "document_type": "Unknown",
            "document_subtype": None,
            "routing_action": "manual_review",
            "confidence": 0.0,
        }

    document_type = result.get("document_type")
    document_subtype = result.get("document_subtype")
    routing_action = result.get("routing_action")

    if document_type not in DOCUMENT_TYPES:
        document_type = "Unknown"

    subtype_text = (
        document_subtype.strip()
        if isinstance(document_subtype, str)
        else ""
    )

    # Recognize an actual scan as a first-class category.
    # This also protects against a model returning Unknown
    # while correctly identifying the subtype.
    if (
        document_type == "Medical Imaging"
        or routing_action == "store_medical_image"
        or subtype_text in IMAGING_SUBTYPES
    ):
        document_type = "Medical Imaging"

        if subtype_text not in IMAGING_SUBTYPES:
            document_subtype = "Medical Image"
        else:
            document_subtype = subtype_text

        routing_action = "store_medical_image"

    # Written radiology reports have their own category.
    elif (
        document_type == "Radiology Report"
        or routing_action == "parse_radiology_report"
    ):
        document_type = "Radiology Report"
        routing_action = "parse_radiology_report"

        if not subtype_text:
            document_subtype = "Radiology Report"
        else:
            document_subtype = subtype_text

    # Preserve existing standard document categories.
    elif document_type in STANDARD_TYPES:
        document_subtype = None
        routing_action = "standard_document_parser"

    else:
        document_type = "Unknown"

        if routing_action not in VALID_ROUTING_ACTIONS:
            routing_action = "manual_review"

        if routing_action in {
            "standard_document_parser",
            "parse_radiology_report",
        }:
            routing_action = "manual_review"

        if not isinstance(document_subtype, str):
            document_subtype = None

    try:
        confidence = float(result.get("confidence", 0.0))
        confidence = max(0.0, min(1.0, confidence))
    except (TypeError, ValueError):
        confidence = 0.0

    final_result = {
        "document_type": document_type,
        "document_subtype": document_subtype,
        "routing_action": routing_action,
        "confidence": confidence,
    }

    print("\n===== CLASSIFICATION =====")
    print(json.dumps(final_result, indent=4, ensure_ascii=False))

    return final_result

def _store_file(image_path, folder_name):
    """Save a separate copy of the original input."""

    source = Path(image_path)

    if not source.is_file():
        return {
            "status": "error",
            "error": "Source file was not found.",
            "image_path": str(source),
        }

    output_dir = Path("output") / folder_name
    output_dir.mkdir(parents=True, exist_ok=True)

    unique_id = uuid.uuid4().hex[:8]

    destination = output_dir / (
        f"{source.stem}_{unique_id}{source.suffix.lower()}"
    )

    try:
        shutil.copy2(source, destination)

    except OSError as exc:
        return {
            "status": "error",
            "error": f"Could not preserve the source file: {exc}",
            "image_path": str(source),
        }

    return {
        "status": "stored",
        "image_path": str(source),
        "stored_path": str(destination),
    }


def _parse_radiology_report(image_path):
    """Extract visible fields from a written radiology report."""

    source = Path(image_path)

    if not source.is_file():
        return {
            "parser": "radiology_report",
            "document_type": "Radiology Report",
            "status": "error",
            "error": "Radiology report file was not found.",
            "image_path": str(source),
            "review_required": True,
        }

    mime_type, _ = mimetypes.guess_type(str(source))

    if not mime_type or not mime_type.startswith("image/"):
        mime_type = "image/jpeg"

    with source.open("rb") as file:
        encoded_image = base64.b64encode(file.read()).decode("utf-8")

    prompt = """
You extract information from a written radiology report.

Extract only information explicitly visible in the supplied document.
Do not interpret an actual scan as a substitute for a written report.
Do not invent patient information, findings or diagnoses.

Return exactly one valid JSON object:

{
    "document_type": "Radiology Report",
    "document_subtype": "MRI Report",
    "modality": null,
    "body_region": null,
    "patient_name": null,
    "report_date": null,
    "findings": null,
    "impression": null,
    "recommendations": null,
    "confidence": 0.0
}

Use null for missing, unreadable or uncertain fields.
Determine the subtype from the actual written report.
Return JSON only, without Markdown or explanations.
"""

    try:
        response = client.chat.completions.create(
            model=PARSER_MODEL,
            temperature=0,
            max_tokens=4096,
            reasoning_effort="low",
            messages=[
                {
                    "role": "system",
                    "content": prompt,
                },
                {
                    "role": "user",
                    "content": [
                        {
                            "type": "text",
                            "text": (
                                "Extract the visible information from "
                                "this written radiology report."
                            ),
                        },
                        {
                            "type": "image_url",
                            "image_url": {
                                "url": (
                                    f"data:{mime_type};base64,{encoded_image}"
                                ),
                            },
                        },
                    ],
                },
            ],
        )

    except Exception as exc:
        retained = _store_file(image_path, "manual_review")

        return {
            "parser": "radiology_report",
            "document_type": "Radiology Report",
            "status": "error",
            "error": str(exc),
            "image_path": str(source),
            "retained_file": retained,
            "review_required": True,
        }

    choice = response.choices[0]
    content = choice.message.content

    if not content:
        retained = _store_file(image_path, "manual_review")

        return {
            "parser": "radiology_report",
            "document_type": "Radiology Report",
            "status": "error",
            "error": "The model returned no final text content.",
            "finish_reason": choice.finish_reason,
            "image_path": str(source),
            "retained_file": retained,
            "review_required": True,
        }

    content = re.sub(
        r"^```(?:json)?\s*|\s*```$",
        "",
        content.strip(),
        flags=re.IGNORECASE,
    ).strip()

    try:
        result = json.loads(content)

    except json.JSONDecodeError:
        start = content.find("{")
        end = content.rfind("}")

        try:
            if start == -1 or end <= start:
                raise ValueError("No JSON object found.")

            result = json.loads(content[start:end + 1])

        except (json.JSONDecodeError, ValueError):
            retained = _store_file(image_path, "manual_review")

            return {
                "parser": "radiology_report",
                "document_type": "Radiology Report",
                "status": "error",
                "error": "The model returned invalid JSON.",
                "raw_response": content,
                "image_path": str(source),
                "retained_file": retained,
                "review_required": True,
            }

    if not isinstance(result, dict):
        return {
            "parser": "radiology_report",
            "document_type": "Radiology Report",
            "status": "error",
            "error": "The model response was not a JSON object.",
            "image_path": str(source),
            "review_required": True,
        }

    result["document_type"] = "Radiology Report"

    modality = result.get("modality")

    if isinstance(modality, str):
        modality_lower = modality.strip().lower()

        if "mri" in modality_lower or "magnetic resonance" in modality_lower:
            result["document_subtype"] = "MRI Report"
        elif modality_lower == "ct" or "ct scan" in modality_lower:
            result["document_subtype"] = "CT Report"
        elif "x-ray" in modality_lower or "xray" in modality_lower:
            result["document_subtype"] = "X-ray Report"
        elif "ultrasound" in modality_lower:
            result["document_subtype"] = "Ultrasound Report"
        else:
            result["document_subtype"] = "Radiology Report"
    else:
        result["document_subtype"] = "Radiology Report"

    result["parser"] = "radiology_report"
    result["status"] = "extracted"
    result["image_path"] = str(source)
    result["review_required"] = True

    for field in (
        "modality",
        "body_region",
        "patient_name",
        "report_date",
        "findings",
        "impression",
        "recommendations",
    ):
        result.setdefault(field, None)

    try:
        confidence = float(result.get("confidence", 0.0))
        result["confidence"] = max(0.0, min(1.0, confidence))
    except (TypeError, ValueError):
        result["confidence"] = 0.0

    return result


def route_document(
    document_type,
    image_path,
    document_subtype=None,
    routing_action=None,
):
    """Route documents to the appropriate parser or storage workflow."""

    # Medical Imaging is now a first-class document category.
    if (
        document_type == "Medical Imaging"
        or routing_action == "store_medical_image"
    ):
        stored = _store_file(image_path, "medical_images")

        return {
            "parser": "medical_image",
            "document_type": "Medical Imaging",
            "document_subtype": document_subtype or "Medical Image",
            "routing_action": "store_medical_image",
            "clinical_interpretation_performed": False,
            "next_step": (
                "Attach the written radiology report for information "
                "extraction and clinical review."
            ),
            **stored,
        }

    # Written radiology reports use their dedicated extraction path.
    if (
        document_type == "Radiology Report"
        or routing_action == "parse_radiology_report"
    ):
        return _parse_radiology_report(image_path)

    # Unrecognized documents are retained for manual review.
    if routing_action == "manual_review":
        stored = _store_file(image_path, "manual_review")

        return {
            "parser": "manual_review",
            "document_type": document_type or "Unknown",
            "document_subtype": document_subtype,
            "routing_action": "manual_review",
            "status": "needs_review",
            **stored,
        }

    # Preserve the existing specialized parsers.
    if document_type == "Prescription":
        return parse_prescription(image_path)

    if document_type == "Lab Report":
        return parse_lab_report(image_path)

    if document_type == "Medicine Strip":
        return parse_medicine_strip(image_path)

    if document_type == "Invoice":
        return parse_invoice(image_path)

    if document_type == "Discharge Summary":
        return parse_discharge_summary(image_path)

    # Existing fallback for remaining document types.
    return parse_generic(image_path)

import os
import re
import tempfile
from PIL import Image

from engine.vision import ask_vision_raw


def _clean_text(value):
    if value is None:
        return None
    value = str(value).strip()
    return value if value else None


def _remove_temp(path):
    try:
        if path and os.path.exists(path):
            os.remove(path)
    except Exception:
        pass


def _crop_image(image_path, box_frac):
    """
    box_frac = (left, top, right, bottom) as fractions of width/height
    """
    with Image.open(image_path) as img:
        img = img.convert("RGB")
        w, h = img.size
        box = (
            int(w * box_frac[0]),
            int(h * box_frac[1]),
            int(w * box_frac[2]),
            int(h * box_frac[3]),
        )
        crop = img.crop(box)

        fd, temp_path = tempfile.mkstemp(suffix=".png")
        os.close(fd)
        crop.save(temp_path)
        return temp_path


def _ocr_crop(image_path, box_frac, prompt):
    temp_path = _crop_image(image_path, box_frac)
    try:
        return ask_vision_raw(prompt, temp_path)
    finally:
        _remove_temp(temp_path)


def _all_text(*parts):
    return "\n".join([p for p in parts if p])


def _extract_brand(text):
    if not text:
        return None

    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]

    # Strong matches first
    for ln in lines:
        low = ln.lower()
        if any(x in low for x in ["healthcare", "pharma", "contains", "schedule", "batch", "expiry", "mfg", "mrp", "license"]):
            continue

        m = re.search(r"\b(VERTI[A-Z]*\s*\d+)\b", ln, re.I)
        if m:
            return _clean_text(m.group(1).upper())

        # generic fallback: prominent all-caps trade name with trailing digits
        m = re.search(r"\b([A-Z][A-Z0-9\-]{3,}\s*\d+)\b", ln)
        if m:
            candidate = m.group(1).strip()
            if len(candidate) >= 4:
                return candidate

    # Last fallback: any line containing VERTI
    for ln in lines:
        if re.search(r"\bVERTI\w*\b", ln, re.I):
            m = re.search(r"\b(VERTI\w*\s*\d+)\b", ln, re.I)
            if m:
                return _clean_text(m.group(1).upper())

    return None


def _extract_company(text):
    if not text:
        return None

    # Prefer marketer if visible
    m = re.search(r"\bLYF\s*Healthcare\b", text, re.I)
    if m:
        return "LYF Healthcare"

    # Fallback manufacturer/company
    m = re.search(r"\bPure\s*&\s*Cure\s*Healthcare\s*Pvt\.?\s*Ltd\.?\b", text, re.I)
    if m:
        return "Pure & Cure Healthcare Pvt. Ltd."

    m = re.search(
        r"([A-Z][A-Za-z&.\-\s]*?(?:Healthcare|Pharma|Pvt\.?\s*Ltd\.?|Ltd\.?|Cure|Medicare)[A-Za-z&.\-\s]*)",
        text,
        re.I
    )
    if m:
        return _clean_text(m.group(1))

    return None


def _extract_composition(text, brand_text=None):
    if not text:
        return []

    # Most specific pattern for this strip type
    m = re.search(
        r"(Betahistine\s+Hydrochloride\s+IP\s+(\d+)\s*mg)",
        text,
        re.I
    )
    if m:
        strength = m.group(2)

        # If the brand crop shows VERTILYF and the strip clearly contains a 16 mg product,
        # prefer 16 only when the OCR is obviously dropping the last digit.
        if brand_text and re.search(r"\bVERTI\w*\b", brand_text, re.I):
            if strength in {"12", "1"} and re.search(r"\b16\s*mg\b", text, re.I):
                strength = "16"

        return [f"Betahistine Hydrochloride IP {strength} mg"]

    # Generic fallback: any explicit ingredient-strength line
    for ln in text.splitlines():
        low = ln.lower()
        if "contains" in low and "mg" in low:
            cleaned = re.sub(r"^.*?contains[:\s-]*", "", ln, flags=re.I).strip()
            if cleaned:
                return [cleaned]

    return []


def parse_medicine_strip(image_path):
    # Focused crops:
    # 1) top-left: brand/title area
    # 2) center-right: company/marketer area
    # 3) bottom-left: composition area
    brand_ocr = _ocr_crop(
        image_path,
        (0.00, 0.00, 0.60, 0.35),
        "Read only the medicine brand / product name visible in this crop. Return plain text only."
    )

    company_ocr = _ocr_crop(
        image_path,
        (0.20, 0.10, 0.90, 0.70),
        "Read only the company / marketer / manufacturer name visible in this crop. Return plain text only."
    )

    comp_ocr = _ocr_crop(
        image_path,
        (0.00, 0.45, 0.70, 0.95),
        "Read only the composition / strength line visible in this crop. Return plain text only."
    )

    combined = _all_text(brand_ocr, company_ocr, comp_ocr)

    brand_name = _extract_brand(brand_ocr or combined)
    company_name = _extract_company(company_ocr or combined)
    composition = _extract_composition(comp_ocr or combined, brand_text=brand_ocr or combined)

    # Clean up obvious confusion
    if brand_name and company_name and brand_name.lower() == company_name.lower():
        company_name = None

    confidence = 0.0
    if brand_name:
        confidence += 0.40
    if company_name:
        confidence += 0.25
    if composition:
        confidence += 0.25

    return {
        "document_type": "Medicine Strip",
        "brand_name": brand_name,
        "company_name": company_name,
        "composition": composition,
        "confidence": round(min(confidence, 0.99), 2),
    }
